cmake_minimum_required(VERSION 3.16)
project(aloetls LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
file(GLOB MBEDTLS_SOURCES CONFIGURE_DEPENDS vendor/library/*.c)
add_library(aloetls SHARED tls_bridge.cpp tcp_bridge.cpp file_identity.cpp ${MBEDTLS_SOURCES})
set_target_properties(aloetls PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
target_include_directories(aloetls PRIVATE vendor/include vendor/library)
target_compile_options(aloetls PRIVATE -O2 -fexceptions)
target_link_libraries(aloetls PUBLIC libace_napi.z.so)
