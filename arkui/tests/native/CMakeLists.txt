cmake_minimum_required(VERSION 3.16)
project(aloetls_host LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../AloeSend/entry/src/main/cpp")
file(GLOB MBEDTLS_SOURCES "${SOURCE_ROOT}/vendor/library/*.c")
add_library(aloetls MODULE "${SOURCE_ROOT}/tls_bridge.cpp" "${SOURCE_ROOT}/tcp_bridge.cpp" ${MBEDTLS_SOURCES})
target_include_directories(aloetls PRIVATE host-include "${NODE_HEADERS}" "${SOURCE_ROOT}/vendor/include" "${SOURCE_ROOT}/vendor/library")
target_compile_options(aloetls PRIVATE -O2 -fexceptions)
target_link_libraries(aloetls PRIVATE "${CMAKE_BINARY_DIR}/libnode.a" ws2_32 bcrypt)
target_link_options(aloetls PRIVATE -static-libgcc -static-libstdc++)
set_target_properties(aloetls PROPERTIES PREFIX "" SUFFIX ".node")
